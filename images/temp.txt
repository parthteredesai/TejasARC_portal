            async function loadSolarData() {
            const res = await fetch("/api/solar"); 
            const data = await res.json();

            const N = 20;
            const recent = data.slice(-N).reverse();

            const labels = recent.map(d => 
                new Date(d.insignt.timestamp).toLocaleTimeString()
            );

            const voltage = recent.map(d => d.sensors.bus_voltage);

            new Chart(document.getElementById("busVoltageChart"), {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: "Bus Voltage (V)",
                        data: voltage,
                        borderWidth: 2,
                        borderColor: "#0d4e87",
                        backgroundColor: "rgba(30,136,229,0.15)",
                        tension: 0.3,
                        fill: true
                    }]
                },
                options: {
                        responsive: true,
                        maintainAspectRatio: false 
                }
            });
        }

        loadSolarData();
        </script>
